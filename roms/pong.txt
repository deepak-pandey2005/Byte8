# Pong - Classic Two-Player / AI Arcade Game
# Screen resolution: 64x32
# Controls:
#   v5, v6: Paddle 1 Y, Paddle 2 Y
#   v7, v8: Ball X, Ball Y
#   v9, vA: Ball Velocity X, Ball Velocity Y
#   vB, vC: Score Player 1, Score Player 2

start:
    cls
    v5 = 12         ; Paddle 1 Y (x=2)
    v6 = 12         ; Paddle 2 Y (x=61)
    vB = 0          ; P1 score
    vC = 0          ; P2 score

reset_ball:
    # Set sound timer to beep on reset/score
    sound = v9
    v7 = 31         ; Ball X center
    v8 = 15         ; Ball Y center
    v9 = 1          ; Vel X (1=right, 0=left)
    vA = 1          ; Vel Y (1=down, 0=up)

draw_all:
    cls
    # Draw left paddle (x=2, y=v5, height=5)
    v0 = 2
    v1 = v5
    i = paddle_sprite
    draw v0, v1, 5

    # Draw right paddle (x=61, y=v6, height=5)
    v0 = 61
    v1 = v6
    i = paddle_sprite
    draw v0, v1, 5

    # Draw ball (x=v7, y=v8, height=2)
    v0 = v7
    v1 = v8
    i = ball_sprite
    draw v0, v1, 2

game_loop:
    # AI logic for right paddle: track ball
    if v6 == v8
    jump check_left_paddle

    v6 += 1
    if v6 != 27
    jump check_left_paddle
    v6 = 26

check_left_paddle:
    # Player 1 paddle tracking
    if v5 == v8
    jump move_ball

    v5 += 1
    if v5 != 27
    jump move_ball
    v5 = 26

move_ball:
    if v9 == 1
    jump ball_moving_right

ball_moving_left:
    v7 -= 1
    if v7 != 3
    jump check_left_wall

    # Paddle 1 collision check
    v0 = v8
    if v0 == v5
    jump bounce_x
    v0 += 1
    if v0 == v5
    jump bounce_x
    v0 += 1
    if v0 == v5
    jump bounce_x

check_left_wall:
    if v7 == 0
    jump score_p2
    jump update_y

ball_moving_right:
    v7 += 1
    if v7 != 60
    jump check_right_wall

    # Paddle 2 collision check
    v0 = v8
    if v0 == v6
    jump bounce_x
    v0 += 1
    if v0 == v6
    jump bounce_x
    v0 += 1
    if v0 == v6
    jump bounce_x

check_right_wall:
    if v7 == 63
    jump score_p1
    jump update_y

bounce_x:
    # Beep buzzer on bounce
    v0 = 2
    sound = v0
    if v9 == 1
    jump set_vx_neg
    v9 = 1
    jump update_y

set_vx_neg:
    v9 = 0
    jump update_y

update_y:
    if vA == 1
    jump ball_down

ball_up:
    v8 -= 1
    if v8 == 0
    vA = 1
    jump render_frame

ball_down:
    v8 += 1
    if v8 == 30
    vA = 0

render_frame:
    jump draw_all

score_p1:
    vB += 1
    v0 = 5
    sound = v0
    jump reset_ball

score_p2:
    vC += 1
    v0 = 5
    sound = v0
    jump reset_ball

# Sprites
paddle_sprite:
    db 0b10000000
    db 0b10000000
    db 0b10000000
    db 0b10000000
    db 0b10000000

ball_sprite:
    db 0b11000000
    db 0b11000000
